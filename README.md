const OpenAI = require("openai")



const openai = new OpenAI({

 apiKey: process.env.OPENAI_KEY

})



async function nvlDecision(prompt){



const completion = await openai.chat.completions.create({

 model:"gpt-4o-mini",

 messages:[

  {role:"system",content:"Você é a NVL AI da Frame & Foco. Seu objetivo é estruturar crescimento empresarial para microempreendedores seguindo o Método NVL."},

  {role:"user",content:prompt}

 ]

})



return completion.choices[0].message.content

}



module.exports = nvlDecision
